    29→type IconKey =
    30→  | "flask" | "syringe" | "droplet" | "armchair" | "activity"
    31→  | "heartpulse" | "video" | "monitor" | "credit" | "badge"
    32→  | "shield" | "megaphone" | "users" | "message" | "building" | "trending"
    33→  | "package";

    35→function iconForCategory(category: string): IconKey {
    36→  const k = category.toLowerCase();
    37→  if (k.includes("lab")) return "flask";
    38→  if (k.includes("diagnostic")) return "activity";
    39→  if (k.includes("injection")) return "syringe";
    40→  if (k.includes("phlebotomy")) return "droplet";
    41→  if (k.includes("furniture")) return "armchair";
    42→  if (k.includes("body-composition") || k.includes("body composition")) return "activity";
    43→  if (k.includes("recovery")) return "heartpulse";
    44→  if (k.includes("telehealth")) return "video";
    45→  if (k.includes("software")) return "monitor";
    46→  if (k.includes("billing")) return "credit";
    47→  if (k.includes("credential")) return "badge";
    48→  if (k.includes("compliance")) return "shield";
    49→  if (k.includes("marketing")) return "megaphone";
    50→  if (k.includes("staffing")) return "users";
    51→  if (k.includes("patient engagement")) return "message";
    52→  if (k.includes("expansion")) return "trending";
    53→  return "package";
    54→}

   464→function QuoteForm({
   465→  listingId,
   466→  listingTitle,
   467→  onDone,
   468→}: {
   469→  listingId: string;
   470→  listingTitle: string;
   471→  onDone?: () => void;
   472→}) {
   473→  const [form, setForm] = useState<QuoteFormState>(emptyForm);
   474→  const [submitting, setSubmitting] = useState(false);
   475→  const [done, setDone] = useState(false);
   476→
   477→  async function submit(e: React.FormEvent) {
   478→    e.preventDefault();
   479→    if (!form.consent) {
   480→      toast.error("Please acknowledge the consent statement before submitting.");
   481→      return;
   482→    }
   483→    setSubmitting(true);
   484→    try {
   485→      const res = await fetch("/api/quote", {
   486→        method: "POST",
   487→        headers: { "Content-Type": "application/json" },
   488→        body: JSON.stringify({
   489→          listingId,
   490→          requesterName: form.requesterName,
   491→          requesterEmail: form.requesterEmail,
   492→          requesterOrg: form.requesterOrg || null,
   493→          quantity: form.quantity || null,
   494→          notes: form.notes || null,
   495→        }),
   496→      });
   497→      if (!res.ok) throw new Error();
   498→      setDone(true);
   499→      toast.success("Quote request sent. The vendor will follow up.");
   500→      onDone?.();
   501→    } catch {
   502→      toast.error("Something went wrong. Please try again.");
   503→    } finally {
   504→      setSubmitting(false);
   505→    }
   506→  }
   507→
   508→  if (done) {
   509→    return (
   510→      <div className="flex flex-col items-center gap-2 py-6 text-center">
   511→        <CheckCircle2 className="h-9 w-9 text-teal-600" />
   512→        <p className="text-sm font-medium">Quote request sent for &ldquo;{listingTitle}&rdquo;</p>
   513→        <p className="text-xs text-muted-foreground">The vendor will reach out using the contact details you provided.</p>
   514→      </div>
   515→    );
   516→  }
   517→
   518→  return (
   519→    <form onSubmit={submit} className="grid gap-3 sm:grid-cols-2">
   520→      <div className="grid gap-1.5">
   521→        <Label htmlFor="qf-name" className="text-xs">Requester name *</Label>
   522→        <Input
   523→          id="qf-name"
   524→          required
   525→          placeholder="Full name"
   526→          value={form.requesterName}
   527→          onChange={(e) => setForm({ ...form, requesterName: e.target.value })}
   528→        />
