    70→function avatarColor(index: number) {
    71→  const key = AVATAR_COLORS[index % AVATAR_COLORS.length];

    75→
    76→function formatComp(min: number | null, max: number | null): string | null {
    77→  const isHourly = (n: number | null) => n != null && n < 1000;
    78→  const hourly = isHourly(min) || isHourly(max);
    79→  const fmt = (n: number) => (hourly ? `$${n}/hr` : `$${Math.round(n / 1000)}k`);
    80→  if (min != null && max != null) return `${fmt(min)} – ${fmt(max)}`;
    81→  if (min != null) return `From ${fmt(min)}`;
    82→  if (max != null) return `Up to ${fmt(max)}`;
    83→  return null;
    84→}

   625→type ApplicationForm = {
   626→  applicantName: string;
   627→  applicantEmail: string;
   628→  applicantPhone: string;
   629→  coverNote: string;
   630→  consent: boolean;
   631→};
   632→
   633→const EMPTY_FORM: ApplicationForm = {
   634→  applicantName: "",
   635→  applicantEmail: "",
   636→  applicantPhone: "",
   637→  coverNote: "",
   638→  consent: false,
   639→};
   640→
   641→function JobApplicationDialog({
   642→  job,
   643→  onClose,
   644→}: {
   645→  job: JobPostingT | null;
   646→  onClose: () => void;
   647→}) {
   648→  const [form, setForm] = useState<ApplicationForm>(EMPTY_FORM);
   649→  const [submitting, setSubmitting] = useState(false);
   650→  const [done, setDone] = useState(false);
   651→
   652→  const licenses = job ? splitCsv(job.requiredLicenses) : [];
   653→  const specialties = job ? splitCsv(job.treatmentSpecialties) : [];
   654→  const comp = job ? formatComp(job.compMin, job.compMax) : null;
   655→
   656→  function close() {
   657→    onClose();
   658→    // Reset after the close animation finishes so users see a clean form next time.
   659→    setTimeout(() => {
   660→      setForm(EMPTY_FORM);
   661→      setDone(false);
   662→      setSubmitting(false);
   663→    }, 200);
   664→  }
   665→
   666→  async function submit(e: React.FormEvent) {
   667→    e.preventDefault();
   668→    if (!job) return;
   669→    if (!form.consent) {
   670→      toast.error("Please acknowledge the platform disclosure to continue.");
   671→      return;
   672→    }
   673→    setSubmitting(true);
   674→    try {
   675→      const res = await fetch("/api/job-application", {
   676→        method: "POST",
   677→        headers: { "Content-Type": "application/json" },
   678→        body: JSON.stringify({
   679→          jobPostingId: job.id,
   680→          applicantName: form.applicantName,
   681→          applicantEmail: form.applicantEmail,
   682→          applicantPhone: form.applicantPhone || null,
   683→          coverNote: form.coverNote || null,
   684→        }),
   685→      });
   686→      if (!res.ok) throw new Error();
   687→      setDone(true);
   688→      toast.success("Application submitted. The clinic will reach out if there is a fit.");
   689→    } catch {
   690→      toast.error("Something went wrong. Please try again.");
   691→    } finally {
   692→      setSubmitting(false);
   693→    }
   694→  }
   695→
   696→  return (
   697→    <Dialog
   698→      open={!!job}
   699→      onOpenChange={(v) => {
   700→        if (!v) close();
   701→      }}
   702→    >
   703→      <DialogContent className="max-h-[92vh] overflow-y-auto sm:max-w-2xl">
   704→        {job && (
   705→          <>
   706→            <DialogHeader>
   707→              <p className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
   708→                {job.clinicName}
   709→              </p>
   710→              <DialogTitle className="text-xl">{job.title}</DialogTitle>
   711→              <DialogDescription className="flex flex-wrap items-center gap-2">
   712→                <span className="flex items-center gap-1">
   713→                  <MapPin className="h-3.5 w-3.5" /> {job.city}, {job.state}
   714→                </span>
   715→                <span className="flex items-center gap-1">
   716→                  <Briefcase className="h-3.5 w-3.5" /> {job.employmentType}
   717→                </span>
   718→                {job.remote && <StatusPill tone="sky">Remote</StatusPill>}
   719→              </DialogDescription>
   720→            </DialogHeader>
   721→
   722→            {/* Job summary */}
   723→            <div className="grid gap-3 rounded-xl border border-border bg-muted/30 p-4 text-sm sm:grid-cols-2">
   724→              {comp && (
   725→                <div className="flex items-center gap-2">
   726→                  <DollarSign className="h-4 w-4 text-emerald-700" />
   727→                  <span className="font-semibold text-emerald-700">{comp}</span>
   728→                </div>
   729→              )}
   730→              {job.schedule && (
   731→                <div className="flex items-center gap-2 text-muted-foreground">
   732→                  <CalendarDays className="h-4 w-4" /> {job.schedule}
   733→                </div>
   734→              )}
   735→              {job.requiredExperience && (
   736→                <div className="flex items-center gap-2 text-muted-foreground">
   737→                  <Clock className="h-4 w-4" /> {job.requiredExperience}
   738→                </div>
   739→              )}
   740→              {job.remote && (
   741→                <div className="flex items-center gap-2 text-muted-foreground">
   742→                  <Video className="h-4 w-4 text-teal-600" /> Remote available
   743→                </div>
   744→              )}
   745→            </div>
   746→
   747→            <p className="text-sm leading-relaxed text-muted-foreground">{job.description}</p>
   748→
   749→            {licenses.length > 0 && (
   750→              <div>
   751→                <p className="mb-1.5 text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
   752→                  Required licenses
   753→                </p>
   754→                <div className="flex flex-wrap gap-1.5">
   755→                  {licenses.map((l) => (
   756→                    <Badge
   757→                      key={l}
   758→                      variant="outline"
   759→                      className="border-border bg-muted/40 text-xs text-foreground/80"
   760→                    >
   761→                      {l}
   762→                    </Badge>
   763→                  ))}
   764→                </div>
   765→              </div>
   766→            )}
   767→
   768→            {specialties.length > 0 && (
   769→              <div>
   770→                <p className="mb-1.5 text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
   771→                  Treatment specialties
   772→                </p>
   773→                <div className="flex flex-wrap gap-1.5">
   774→                  {specialties.map((s) => (
   775→                    <Badge
   776→                      key={s}
   777→                      variant="outline"
   778→                      className="border-teal-200 bg-teal-50/50 text-xs text-teal-700"
   779→                    >
   780→                      {s}
   781→                    </Badge>
   782→                  ))}
   783→                </div>
   784→              </div>
   785→            )}
   786→
   787→            {/* Application form / success */}
   788→            <div className="rounded-xl border border-teal-200 bg-teal-50/30 p-5">
   789→              <div className="flex items-center gap-2">
   790→                <ListChecks className="h-4 w-4 text-teal-700" />
   791→                <h4 className="text-sm font-semibold text-foreground">Apply for this role</h4>
   792→              </div>

   794→                Your application is sent to the hiring clinic. Novalyte AI is a
   795→                technology platform and does not employ or supervise providers.
   796→              </p>

   798→              {!done ? (

   825→                        setForm({ ...form, applicantEmail: e.target.value })
   826→                      }
   827→                    />
   828→                  </div>
