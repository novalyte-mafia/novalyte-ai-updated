    65→/* Category icon — stable wrapper (switch with literal JSX so the React

    67→function CategoryIcon({ category, className }: { category: string; className?: string }) {
    68→  switch (category) {
    69→    case "Laboratory Services": return <FlaskConical className={className} />;
    70→    case "Diagnostic Equipment": return <Activity className={className} />;
    71→    case "Injection Supplies": return <Syringe className={className} />;
    72→    case "Phlebotomy Supplies": return <Droplet className={className} />;
    73→    case "Medical Furniture": return <Armchair className={className} />;
    74→    case "Body-Composition Systems": return <Stethoscope className={className} />;
    75→    case "Recovery Technology": return <HeartPulse className={className} />;
    76→    case "Telehealth Tools": return <Video className={className} />;
    77→    case "Clinic Software": return <Monitor className={className} />;
    78→    case "Billing Services": return <CreditCard className={className} />;
    79→    case "Credentialing Services": return <BadgeCheck className={className} />;
    80→    case "Compliance Support": return <ShieldCheck className={className} />;
    81→    case "Marketing Services": return <Megaphone className={className} />;
    82→    case "Staffing Services": return <Users className={className} />;
    83→    case "Patient Engagement Tools": return <MessageSquare className={className} />;
    84→    case "Clinic Expansion Services": return <Building2 className={className} />;
    85→    default: return <Package className={className} />;
    86→  }
    87→}

    89→function availabilityMeta(av: string): { tone: "teal" | "amber" | "sky" | "violet"; label: string } {
    90→  const a = av.toLowerCase();
    91→  if (a.includes("order")) return { tone: "amber", label: "Made to order" };
    92→  if (a.includes("limit")) return { tone: "violet", label: "Limited" };
    93→  if (a.includes("pre") || a.includes("back")) return { tone: "sky", label: "Pre-order" };
    94→  return { tone: "teal", label: "In stock" };
    95→}

   594→/* ───────────────────────────────────────────────────────────────
   595→   Inline quote form (used inside the vendor listing card dialog)
   596→   ─────────────────────────────────────────────────────────────── */
   597→function QuoteInlineForm({ listing, onClose }: { listing: MarketplaceListingT; onClose: () => void }) {
   598→  const [form, setForm] = useState({
   599→    requesterName: "",
   600→    requesterEmail: "",
   601→    requesterOrg: "",
   602→    quantity: "",
   603→    notes: "",
   604→    consent: false,
   605→  });
   606→  const [submitting, setSubmitting] = useState(false);
   607→  const [done, setDone] = useState(false);
   608→
   609→  async function submit(e: React.FormEvent) {
   610→    e.preventDefault();
   611→    if (!form.consent) {
   612→      toast.error("Please acknowledge the platform terms before submitting.");
   613→      return;
   614→    }
   615→    setSubmitting(true);
   616→    try {
   617→      const res = await fetch("/api/quote", {
   618→        method: "POST",
   619→        headers: { "Content-Type": "application/json" },
   620→        body: JSON.stringify({
   621→          listingId: listing.id,
   622→          requesterName: form.requesterName,
   623→          requesterEmail: form.requesterEmail,
   624→          requesterOrg: form.requesterOrg || null,
   625→          quantity: form.quantity || null,
   626→          notes: form.notes || null,
   627→        }),
   628→      });
   629→      if (!res.ok) throw new Error("failed");
   630→      setDone(true);
   631→      toast.success("Quote request submitted", {
   632→        description: `${listing.vendorName} will follow up via email.`,
   633→      });
   634→    } catch {
   635→      toast.error("Something went wrong. Please try again.");
   636→    } finally {
   637→      setSubmitting(false);
   638→    }
   639→  }
   640→
   641→  if (done) {
   642→    return (
   643→      <div className="flex flex-col items-center gap-2 py-6 text-center">
   644→        <span className="flex h-12 w-12 items-center justify-center rounded-full bg-teal-50">
   645→          <CheckCircle2 className="h-6 w-6 text-teal-600" />
   646→        </span>
   647→        <p className="text-base font-semibold text-foreground">Quote request submitted</p>
   648→        <p className="max-w-sm text-sm text-muted-foreground">
   649→          We routed your inquiry to {listing.vendorName}. Expect a follow-up at the email you provided.
   650→        </p>
   651→        <Button variant="outline" size="sm" onClick={onClose} className="mt-2">Close</Button>
   652→      </div>
   653→    );
   654→  }
   655→
   656→  return (
   657→    <form onSubmit={submit} className="space-y-3">
   658→      <div className="grid gap-3 sm:grid-cols-2">
   659→        <div className="grid gap-1.5">
   660→          <Label className="text-xs">Full name <span className="text-rose-500">*</span></Label>
   661→          <Input required value={form.requesterName} onChange={(e) => setForm({ ...form, requesterName: e.target.value })} placeholder="Your name" />
   662→        </div>
   663→        <div className="grid gap-1.5">
   664→          <Label className="text-xs">Work email <span className="text-rose-500">*</span></Label>
   665→          <Input required type="email" value={form.requesterEmail} onChange={(e) => setForm({ ...form, requesterEmail: e.target.value })} placeholder="you@clinic.com" />
   666→        </div>
   667→      </div>
   668→      <div className="grid gap-1.5">
   669→        <Label className="text-xs">Organization / clinic</Label>
   670→        <Input value={form.requesterOrg} onChange={(e) => setForm({ ...form, requesterOrg: e.target.value })} placeholder="Clinic or company name" />
   671→      </div>
   672→      <div className="grid gap-3 sm:grid-cols-2">
   673→        <div className="grid gap-1.5">
   674→          <Label className="text-xs">Quantity / scope</Label>
   675→          <Input value={form.quantity} onChange={(e) => setForm({ ...form, quantity: e.target.value })} placeholder="e.g. 10 units" />
   676→        </div>
   677→        <div className="grid gap-1.5">
   678→          <Label className="text-xs">Notes</Label>
   679→          <Textarea rows={2} value={form.notes} onChange={(e) => setForm({ ...form, notes: e.target.value })} placeholder="Timeline, requirements" className="resize-none" />
   680→        </div>
   681→      </div>
   682→      <label className="flex items-start gap-2 text-xs text-muted-foreground">
   683→        <input type="checkbox" checked={form.consent} onChange={(e) => setForm({ ...form, consent: e.target.checked })} className="mt-0.5" />
   684→        <span>
   685→          I understand Novalyte AI is a technology platform that facilitates commerce inquiries and
   686→          does not sell, warranty, or fulfill products directly.
   687→        </span>
   688→      </label>
   689→      <Button type="submit" className="w-full bg-teal-600 text-white hover:bg-teal-700" disabled={submitting}>
   690→        {submitting ? "Sending…" : <>Send quote request <ArrowRight className="ml-1 h-4 w-4" /></>}
   691→      </Button>
   692→    </form>
   693→  );

   696→/* ───────────────────────────────────────────────────────────────
   697→   Contact form (POSTs to /api/contact with role="vendor")

   699→function ContactForm({
   700→  vendor,
   701→  defaultListingId,
   702→  onSuccess,
   703→}: {
   704→  vendor: VendorT;
   705→  defaultListingId?: string;
   706→  onSuccess?: () => void;
   707→}) {
   708→  const [form, setForm] = useState({
   709→    name: "",
   710→    email: "",
   711→    organization: "",
   712→    message: "",
   713→    consent: false,
   714→  });
   715→  const [submitting, setSubmitting] = useState(false);
   716→  const [done, setDone] = useState(false);
   717→
   718→  async function submit(e: React.FormEvent) {
   719→    e.preventDefault();
   720→    if (!form.consent) {
   721→      toast.error("Please acknowledge the platform terms before submitting.");
   722→      return;
   723→    }
   724→    setSubmitting(true);
   725→    try {
   726→      // Compose a vendor-context message that includes a quote/listing reference if available.
   727→      const context = defaultListingId
   728→        ? `[Novalyte marketplace — vendor: ${vendor.name} — listing ref: ${defaultListingId}] ${form.message}`
   729→        : `[Novalyte marketplace — vendor: ${vendor.name}] ${form.message}`;
   730→      const payload: { name: string; email: string; role: "vendor"; message: string } = {
   731→        name: form.name,
   732→        email: form.email,
   733→        role: "vendor",
   734→        message: form.organization ? `[Org: ${form.organization}] ${context}` : context,
   735→      };
   736→      const res = await fetch("/api/contact", {
   737→        method: "POST",
   738→        headers: { "Content-Type": "application/json" },
   739→        body: JSON.stringify(payload),
   740→      });
   741→      if (!res.ok) throw new Error("failed");
   742→      setDone(true);
   743→      toast.success("Message sent", {
   744→        description: `Your inquiry has been routed to ${vendor.name}.`,
   745→      });

   747→    } catch {
   748→      toast.error("Something went wrong. Please try again.");
   749→    } finally {
   750→      setSubmitting(false);
   751→    }
   752→  }

   755→    return (
   756→      <div className="flex flex-col items-center gap-2 py-8 text-center">

   758→          <CheckCircle2 className="h-6 w-6 text-teal-600" />
   759→        </span>
   760→        <p className="text-base font-semibold text-foreground">Message sent</p>
   761→        <p className="max-w-sm text-sm text-muted-foreground">
   762→          Your inquiry has been routed to {vendor.name}. Expect a follow-up at the email you provided.
   763→        </p>
   764→      </div>
   765→    );
   766→  }
   767→
   768→  return (
   769→    <form onSubmit={submit} className="space-y-3">
   770→      <div className="grid gap-3 sm:grid-cols-2">
   771→        <div className="grid gap-1.5">
   772→          <Label className="text-xs">Full name <span className="text-rose-500">*</span></Label>
   773→          <Input required value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="Your name" />
   774→        </div>
   775→        <div className="grid gap-1.5">
   776→          <Label className="text-xs">Work email <span className="text-rose-500">*</span></Label>
   777→          <Input required type="email" value={form.email} onChange={(e) => setForm({ ...form, email: e.target.value })} placeholder="you@clinic.com" />
   778→        </div>
   779→      </div>
   780→      <div className="grid gap-1.5">
   781→        <Label className="text-xs">Organization / clinic</Label>
   782→        <Input value={form.organization} onChange={(e) => setForm({ ...form, organization: e.target.value })} placeholder="Clinic or company name" />
   783→      </div>
   784→      <div className="grid gap-1.5">
   785→        <Label className="text-xs">Message <span className="text-rose-500">*</span></Label>
   786→        <Textarea
   787→          required
   788→          rows={5}
   789→          value={form.message}

   792→        />
   793→      </div>
   794→      <label className="flex items-start gap-2 text-xs text-muted-foreground">
   795→        <input type="checkbox" checked={form.consent} onChange={(e) => setForm({ ...form, consent: e.target.checked })} className="mt-0.5" />
   796→        <span>
   797→          I understand Novalyte AI is a technology platform that facilitates commerce inquiries and
   798→          does not sell, warranty, or fulfill products directly. Verification is not an endorsement.
   799→        </span>
   800→      </label>
   801→      <Button type="submit" className="w-full bg-teal-600 text-white hover:bg-teal-700" disabled={submitting}>
   802→        {submitting ? "Sending…" : <>Send message <ArrowRight className="ml-1 h-4 w-4" /></>}
   803→      </Button>
