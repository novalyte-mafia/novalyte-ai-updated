   108→function formatComp(min: number | null, max: number | null): string | null {
   109→  const isHourly = (n: number | null) => n != null && n < 1000;
   110→  const hourly = isHourly(min) || isHourly(max);
   111→  const fmt = (n: number) => (hourly ? `$${n}/hr` : `$${Math.round(n / 1000)}k`);
   112→  if (min != null && max != null && min > 0 && max > 0) {
   113→    return `${fmt(min)}\u2013${fmt(max)}`;
   114→  }
   115→  if (min != null && min > 0) return `From ${fmt(min)}`;
   116→  if (max != null && max > 0) return `Up to ${fmt(max)}`;
   117→  return null;
   118→}

   141→function inCompRange(min: number | null, max: number | null, range: string): boolean {
   142→  if (range === "all") return true;
   143→  // Job with no compensation data is excluded when a range is selected.
   144→  if (min == null && max == null) return false;
   145→  const jMin = min ?? 0;
   146→  const jMax = max ?? Number.POSITIVE_INFINITY;
   147→  const [loStr, hiStr] = range.split("-");
   148→  const lo = Number(loStr) * 1000;
   149→  const hi = hiStr === "" ? Number.POSITIVE_INFINITY : Number(hiStr) * 1000;
   150→  return jMin <= hi && jMax >= lo;
   151→}

   325→type JobFilters = {

   335→const DEFAULT_JOB_FILTERS: JobFilters = {

   338→  empType: "all",

   344→
   345→type JobSort = "relevant" | "newest" | "comp-high" | "comp-low";
   346→
   347→const JOB_SORTS: { value: JobSort; label: string }[] = [
   348→  { value: "relevant", label: "Most relevant" },
   349→  { value: "newest", label: "Newest" },
   350→  { value: "comp-high", label: "Compensation: high to low" },
   351→  { value: "comp-low", label: "Compensation: low to high" },

   353→
   354→function JobsPanel({ jobs }: { jobs: JobPostingT[] }) {
   355→  const [filters, setFilters] = useState<JobFilters>(DEFAULT_JOB_FILTERS);
   356→  const [sort, setSort] = useState<JobSort>("relevant");
   357→  const [view, setView] = useState<"grid" | "list">("grid");
   358→  const [page, setPage] = useState(1);
   359→  const [loading, setLoading] = useState(false);
   360→  const [sheetOpen, setSheetOpen] = useState(false);
   361→
   362→  // 300ms simulated loading skeleton — clears via setTimeout when
   363→  // filters/sort/view change. setLoading(true) is fired synchronously
   364→  // in the change handlers below; the effect schedules the clear.
   365→  useEffect(() => {
   366→    const t = window.setTimeout(() => setLoading(false), 300);
   367→    return () => window.clearTimeout(t);
   368→  }, [filters, sort, view]);
   369→
   370→  function applyFilter(patch: Partial<JobFilters>) {
   371→    setFilters((f) => ({ ...f, ...patch }));
   372→    setPage(1);
   373→    setLoading(true);
   374→  }
   375→
   376→  function applySort(v: JobSort) {
   377→    setSort(v);
   378→    setPage(1);
   379→    setLoading(true);
   380→  }
   381→
   382→  function applyView(v: "grid" | "list") {
   383→    setView(v);
   384→    setLoading(true);
   385→  }
   386→
   387→  function clearAll() {
   388→    setFilters(DEFAULT_JOB_FILTERS);
   389→    setPage(1);
   390→    setLoading(true);
   391→  }
   392→
   393→  const update = applyFilter;
   394→
   395→  // Derived filter options
   396→  const allEmpTypes = useMemo(() => {
   397→    const set = new Set<string>();
   398→    jobs.forEach((j) => set.add(j.employmentType));
   399→    return Array.from(set).sort();
   400→  }, [jobs]);
   401→
   402→  const allLicenses = useMemo(() => {
   403→    const set = new Set<string>();
   404→    jobs.forEach((j) => splitCsv(j.requiredLicenses).forEach((l) => set.add(l)));
   405→    return Array.from(set).sort();
   406→  }, [jobs]);
   407→
   408→  const allSpecialties = useMemo(() => {
   409→    const set = new Set<string>();
   410→    jobs.forEach((j) => splitCsv(j.treatmentSpecialties).forEach((s) => set.add(s)));
   411→    return Array.from(set).sort();
   412→  }, [jobs]);
   413→
   414→  // Filtered + sorted results
   415→  const filtered = useMemo(() => {
   416→    const q = filters.query.trim().toLowerCase();
   417→    const out = jobs.filter((j) => {
   418→      if (q) {
   419→        const hay = `${j.title} ${j.clinicName} ${j.city} ${j.state} ${j.description} ${j.treatmentSpecialties ?? ""}`.toLowerCase();
   420→        if (!hay.includes(q)) return false;
   421→      }
   422→      if (filters.state !== "all" && j.state !== filters.state) return false;
   423→      if (filters.empType !== "all" && j.employmentType !== filters.empType) return false;
   424→      if (filters.remoteOnly && !j.remote) return false;
   425→      if (!inCompRange(j.compMin, j.compMax, filters.compRange)) return false;
   426→      if (filters.license !== "all") {
   427→        const ls = splitCsv(j.requiredLicenses);
   428→        if (!ls.includes(filters.license)) return false;
   429→      }
   430→      if (filters.specialty !== "all") {
   431→        const ss = splitCsv(j.treatmentSpecialties);
   432→        if (!ss.includes(filters.specialty)) return false;
   433→      }
   434→      return true;
   435→    });
   436→
   437→    // Sort
   438→    const sorted = [...out];
   439→    if (sort === "comp-high") {
   440→      sorted.sort((a, b) => (b.compMax ?? b.compMin ?? 0) - (a.compMax ?? a.compMin ?? 0));
   441→    } else if (sort === "comp-low") {
   442→      sorted.sort((a, b) => (a.compMin ?? a.compMax ?? Number.POSITIVE_INFINITY) - (b.compMin ?? b.compMax ?? Number.POSITIVE_INFINITY));
   443→    }
   444→    // "newest" and "relevant" both use the natural order from the DB (createdAt desc)
   445→    return sorted;
   446→  }, [jobs, filters, sort]);
   447→
   448→  // Pagination
   449→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   450→  const currentPage = Math.min(page, totalPages);
   451→  const pageItems = filtered.slice(
   452→    (currentPage - 1) * PAGE_SIZE,
   453→    currentPage * PAGE_SIZE,
   454→  );
   455→
   456→  // Active filter chips
   457→  const chips = useMemo(() => {
   458→    const arr: { label: string; clear: () => void }[] = [];
   459→    if (filters.query.trim()) {
   460→      arr.push({ label: `“${filters.query.trim()}”`, clear: () => applyFilter({ query: "" }) });
   461→    }
   462→    if (filters.state !== "all") {
   463→      arr.push({ label: `State: ${filters.state}`, clear: () => applyFilter({ state: "all" }) });
   464→    }
   465→    if (filters.empType !== "all") {
   466→      arr.push({ label: filters.empType, clear: () => applyFilter({ empType: "all" }) });
   467→    }
   468→    if (filters.remoteOnly) {
   469→      arr.push({ label: "Remote only", clear: () => applyFilter({ remoteOnly: false }) });
   470→    }
   471→    if (filters.compRange !== "all") {
   472→      const r = COMP_RANGES.find((c) => c.value === filters.compRange);
   473→      arr.push({ label: r?.label ?? "Comp", clear: () => applyFilter({ compRange: "all" }) });
   474→    }
   475→    if (filters.license !== "all") {
   476→      arr.push({ label: `License: ${filters.license}`, clear: () => applyFilter({ license: "all" }) });
   477→    }
   478→    if (filters.specialty !== "all") {
   479→      arr.push({ label: filters.specialty, clear: () => applyFilter({ specialty: "all" }) });
   480→    }
   481→    return arr;
   482→  }, [filters]);
   483→
   484→  const filtersNode = (
   485→    <JobFiltersContent
   486→      filters={filters}
   487→      update={update}
   488→      allEmpTypes={allEmpTypes}
   489→      allLicenses={allLicenses}
   490→      allSpecialties={allSpecialties}
   491→    />
   492→  );
   493→
   494→  return (
   495→    <div className="grid gap-6 lg:grid-cols-[280px_1fr]">
   496→      {/* Desktop sticky sidebar */}
   497→      <aside className="hidden lg:block">
   498→        <div className="sticky top-20 max-h-[calc(100vh-6rem)] overflow-y-auto novalyte-scroll rounded-2xl border border-border bg-card p-5 shadow-premium-sm">
   499→          <div className="mb-4 flex items-center justify-between">
   500→            <h3 className="text-sm font-semibold text-foreground">Filters</h3>
   501→            {chips.length > 0 && (
   502→              <Button variant="ghost" size="sm" className="h-7 px-2 text-xs" onClick={clearAll}>
   503→                Clear all
   504→              </Button>
   505→            )}
   506→          </div>
   507→          {filtersNode}
   508→        </div>
   509→      </aside>
   510→
   511→      <div>
   512→        {/* Mobile filters + Sort + View toggle row */}
   513→        <div className="mb-4 flex flex-wrap items-center gap-2">
   514→          <Sheet open={sheetOpen} onOpenChange={setSheetOpen}>
   515→            <SheetTrigger asChild>
   516→              <Button variant="outline" size="sm" className="gap-1.5">
   517→                <SlidersHorizontal className="h-4 w-4" /> Filters
   518→                {chips.length > 0 && (
   519→                  <span className="ml-1 inline-flex h-5 min-w-5 items-center justify-center rounded-full bg-teal-600 px-1.5 text-[11px] font-semibold text-white">
   520→                    {chips.length}
   521→                  </span>
   522→                )}
   523→              </Button>
   524→            </SheetTrigger>
   525→            <SheetContent side="left" className="w-[88%] max-w-sm overflow-y-auto">
   526→              <SheetHeader>
   527→                <SheetTitle className="flex items-center gap-2">
   528→                  <SlidersHorizontal className="h-4 w-4" /> Filters
   529→                </SheetTitle>
   530→              </SheetHeader>
   531→              <div className="px-4 pb-6">
   532→                {filtersNode}
   533→                <div className="mt-5 flex gap-2">
   534→                  <Button variant="outline" className="flex-1" onClick={clearAll}>
   535→                    Clear all
   536→                  </Button>
   537→                  <Button
   538→                    className="flex-1 bg-teal-600 text-white hover:bg-teal-700"
   539→                    onClick={() => setSheetOpen(false)}
   540→                  >
   541→                    Show {filtered.length} results
   542→                  </Button>
   543→                </div>
   544→              </div>
   545→            </SheetContent>
   546→          </Sheet>

   717→function buildPageList(current: number, total: number): (number | "…")[] {
   718→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1);
   719→  const out: (number | "…")[] = [1];
   720→  const start = Math.max(2, current - 1);
   721→  const end = Math.min(total - 1, current + 1);
   722→  if (start > 2) out.push("…");
   723→  for (let i = start; i <= end; i++) out.push(i);
   724→  if (end < total - 1) out.push("…");
   725→  out.push(total);
   726→  return out;
   727→}
