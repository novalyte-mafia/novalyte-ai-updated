    29→    29→type IconKey =
    30→    30→  | "flask" | "syringe" | "droplet" | "armchair" | "activity"
    31→    31→  | "heartpulse" | "video" | "monitor" | "credit" | "badge"

    35→    35→function iconForCategory(category: string): IconKey {
    36→    36→  const k = category.toLowerCase();
    37→    37→  if (k.includes("lab")) return "flask";
    38→    38→  if (k.includes("diagnostic")) return "activity";
    39→    39→  if (k.includes("injection")) return "syringe";
    40→    40→  if (k.includes("phlebotomy")) return "droplet";
    41→    41→  if (k.includes("furniture")) return "armchair";
    42→    42→  if (k.includes("body-composition") || k.includes("body composition")) return "activity";
    43→    43→  if (k.includes("recovery")) return "heartpulse";
    44→    44→  if (k.includes("telehealth")) return "video";
    45→    45→  if (k.includes("software")) return "monitor";
    46→    46→  if (k.includes("billing")) return "credit";
    47→    47→  if (k.includes("credential")) return "badge";
    48→    48→  if (k.includes("compliance")) return "shield";
    49→    49→  if (k.includes("marketing")) return "megaphone";
    50→    50→  if (k.includes("staffing")) return "users";
    51→    51→  if (k.includes("patient engagement")) return "message";
    52→    52→  if (k.includes("expansion")) return "trending";
    53→    53→  return "package";
    54→    54→}

   464→   464→function QuoteForm({
   465→   465→  listingId,
   466→   466→  listingTitle,
   467→   467→  onDone,
   468→   468→}: {
   469→   469→  listingId: string;
   470→   470→  listingTitle: string;
   471→   471→  onDone?: () => void;
   472→   472→}) {
   473→   473→  const [form, setForm] = useState<QuoteFormState>(emptyForm);
   474→   474→  const [submitting, setSubmitting] = useState(false);
   475→   475→  const [done, setDone] = useState(false);
   476→   476→
   477→   477→  async function submit(e: React.FormEvent) {
   478→   478→    e.preventDefault();
   479→   479→    if (!form.consent) {
   480→   480→      toast.error("Please acknowledge the consent statement before submitting.");
   481→   481→      return;
   482→   482→    }
   483→   483→    setSubmitting(true);
   484→   484→    try {
   485→   485→      const res = await fetch("/api/quote", {
   486→   486→        method: "POST",
   487→   487→        headers: { "Content-Type": "application/json" },
   488→   488→        body: JSON.stringify({
   489→   489→          listingId,
   490→   490→          requesterName: form.requesterName,
   491→   491→          requesterEmail: form.requesterEmail,
   492→   492→          requesterOrg: form.requesterOrg || null,
   493→   493→          quantity: form.quantity || null,
   494→   494→          notes: form.notes || null,
   495→   495→        }),
   496→   496→      });
   497→   497→      if (!res.ok) throw new Error();
   498→   498→      setDone(true);
   499→   499→      toast.success("Quote request sent. The vendor will follow up.");
   500→   500→      onDone?.();
   501→   501→    } catch {
   502→   502→      toast.error("Something went wrong. Please try again.");
   503→   503→    } finally {
   504→   504→      setSubmitting(false);
   505→   505→    }
   506→   506→  }
   507→   507→
   508→   508→  if (done) {
   509→   509→    return (
   510→   510→      <div className="flex flex-col items-center gap-2 py-6 text-center">
   511→   511→        <CheckCircle2 className="h-9 w-9 text-teal-600" />
   512→   512→        <p className="text-sm font-medium">Quote request sent for &ldquo;{listingTitle}&rdquo;</p>
   513→   513→        <p className="text-xs text-muted-foreground">The vendor will reach out using the contact details you provided.</p>
   514→   514→      </div>
   515→   515→    );
   516→   516→  }
   517→   517→
   518→   518→  return (
   519→   519→    <form onSubmit={submit} className="grid gap-3 sm:grid-cols-2">
   520→   520→      <div className="grid gap-1.5">
   521→   521→        <Label htmlFor="qf-name" className="text-xs">Requester name *</Label>
   522→   522→        <Input
   523→   523→          id="qf-name"
   524→   524→          required
   525→   525→          placeholder="Full name"
   526→   526→          value={form.requesterName}
   527→   527→          onChange={(e) => setForm({ ...form, requesterName: e.target.value })}
   528→   528→        />
   529→   529→      </div>
   530→   530→      <div className="grid gap-1.5">
   531→   531→        <Label htmlFor="qf-email" className="text-xs">Email *</Label>
   532→   532→        <Input
   533→   533→          id="qf-email"
   534→   534→          required
   535→   535→          type="email"
   536→   536→          placeholder="you@clinic.com"
   537→   537→          value={form.requesterEmail}
   538→   538→          onChange={(e) => setForm({ ...form, requesterEmail: e.target.value })}
   539→   539→        />
   540→   540→      </div>
   541→   541→      <div className="grid gap-1.5">
   542→   542→        <Label htmlFor="qf-org" className="text-xs">Organization (optional)</Label>
   543→   543→        <Input
   544→   544→          id="qf-org"
   545→   545→          placeholder="Clinic or company"
   546→   546→          value={form.requesterOrg}
   547→   547→          onChange={(e) => setForm({ ...form, requesterOrg: e.target.value })}
   548→   548→        />
   549→   549→      </div>
   550→   550→      <div className="grid gap-1.5">
   551→   551→        <Label htmlFor="qf-qty" className="text-xs">Quantity (optional)</Label>
   552→   552→        <Input
   553→   553→          id="qf-qty"
   554→   554→          placeholder="e.g. 5 units, 12 months"
   555→   555→          value={form.quantity}
   556→   556→          onChange={(e) => setForm({ ...form, quantity: e.target.value })}
   557→   557→        />
   558→   558→      </div>
   559→   559→      <div className="grid gap-1.5 sm:col-span-2">
   560→   560→        <Label htmlFor="qf-notes" className="text-xs">Notes (optional)</Label>
   561→   561→        <Textarea
   562→   562→          id="qf-notes"
   563→   563→          rows={3}
   564→   564→          placeholder="Questions, delivery timelines, integrations, etc."
   565→   565→          value={form.notes}
   566→   566→          onChange={(e) => setForm({ ...form, notes: e.target.value })}
   567→   567→        />
   568→   568→      </div>
   569→   569→      <label className="flex items-start gap-2 text-xs text-muted-foreground sm:col-span-2">
   570→   570→        <Checkbox
   571→   571→          checked={form.consent}
   572→   572→          onCheckedChange={(v) => setForm({ ...form, consent: v === true })}
   573→   573→          className="mt-0.5"
   574→   574→        />
   575→   575→        <span>
   576→   576→          I understand Novalyte AI is a technology platform and does not sell, endorse, or warrant
   577→   577→          vendor products. Quotes are provided directly by the vendor. Medical or clinical claims
   578→   578→          are subject to moderation.
   579→   579→        </span>
   580→   580→      </label>
   581→   581→      <Button
   582→   582→        type="submit"
   583→   583→        className="sm:col-span-2 bg-teal-600 text-white hover:bg-teal-700"
   584→   584→        disabled={submitting || !form.consent}
   585→   585→      >
   586→   586→        {submitting ? "Sending..." : "Submit quote request"} <ArrowRight className="ml-1 h-4 w-4" />
   587→   587→      </Button>
   588→   588→    </form>
   589→   589→  );
   590→   590→}
