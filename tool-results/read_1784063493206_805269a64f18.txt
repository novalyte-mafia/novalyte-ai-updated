    97→function availabilityMeta(av: string): { tone: "teal" | "amber" | "sky" | "violet"; label: string } {
    98→  const a = av.toLowerCase();
    99→  if (a.includes("order")) return { tone: "amber", label: "Made to order" };
   100→  if (a.includes("limit")) return { tone: "violet", label: "Limited" };
   101→  if (a.includes("pre") || a.includes("back")) return { tone: "sky", label: "Pre-order" };
   102→  return { tone: "teal", label: "In stock" };
   103→}

   629→function pricingModelExplanation(model: string | null): string {
   630→  switch ((model ?? "").toLowerCase()) {
   631→    case "one-time":
   632→      return "A single upfront price for the product or service. No recurring charges unless add-ons are negotiated separately with the vendor.";
   633→    case "subscription":
   634→      return "Recurring billing (monthly or annual) for continued access to the product, software, or service. Pricing tiers may scale with usage, seats, or locations.";
   635→    case "quote":
   636→      return "Pricing is configured per inquiry based on quantity, scope, contract length, and customization. Submit a quote request to receive a tailored proposal.";
   637→    case "range":
   638→      return "Pricing falls within a published range. Final pricing depends on configuration, quantity, and contract terms confirmed during the quote process.";
   639→    case "per-test":
   640→      return "Priced per unit of service (e.g. per lab panel or per draw). Useful for variable-volume clinic workflows; volume tiers may apply.";
   641→    case "percentage":
   642→      return "Priced as a percentage of collections, revenue, or transaction volume. Common for billing, RCM, and revenue-share service arrangements.";
   643→    default:
   644→      return "Pricing details are confirmed with the vendor during the quote process.";
   645→  }
   646→}

   648→/* ───────────────────────────────────────────────────────────────
   649→   Compact quote form (used inline on the pricing card)
   650→   ─────────────────────────────────────────────────────────────── */
   651→function CompactQuoteForm({
   652→  listing,
   653→  onSuccess,
   654→}: {
   655→  listing: MarketplaceListingT;
   656→  onSuccess: () => void;
   657→}) {
   658→  const [form, setForm] = useState({
   659→    requesterName: "",
   660→    requesterEmail: "",
   661→    requesterOrg: "",
   662→    quantity: "",
   663→    notes: "",
   664→    consent: false,
   665→  });
   666→  const [submitting, setSubmitting] = useState(false);
   667→  const [done, setDone] = useState(false);
   668→
   669→  async function submit(e: React.FormEvent) {
   670→    e.preventDefault();
   671→    if (!form.consent) {
   672→      toast.error("Please acknowledge the platform terms before submitting.");
   673→      return;
   674→    }
   675→    setSubmitting(true);
   676→    try {
   677→      const res = await fetch("/api/quote", {
   678→        method: "POST",
   679→        headers: { "Content-Type": "application/json" },
   680→        body: JSON.stringify({
   681→          listingId: listing.id,
   682→          requesterName: form.requesterName,
   683→          requesterEmail: form.requesterEmail,
   684→          requesterOrg: form.requesterOrg || null,
   685→          quantity: form.quantity || null,
   686→          notes: form.notes || null,
   687→        }),
   688→      });
   689→      if (!res.ok) throw new Error("failed");
   690→      setDone(true);
   691→      onSuccess();
   692→    } catch {
   693→      toast.error("Something went wrong. Please try again.");
   694→    } finally {
   695→      setSubmitting(false);
   696→    }
   697→  }
   698→
   699→  if (done) {
   700→    return (
   701→      <div className="mt-4 flex flex-col items-center gap-2 rounded-xl border border-teal-200 bg-teal-50/50 p-5 text-center">
   702→        <CheckCircle2 className="h-7 w-7 text-teal-600" />
   703→        <p className="text-sm font-semibold text-foreground">Quote request submitted</p>
   704→        <p className="max-w-xs text-xs text-muted-foreground">
   705→          We routed your inquiry to {listing.vendorName}. Expect a follow-up at the email you provided.
   706→        </p>
   707→      </div>
   708→    );
   709→  }
   710→
   711→  return (
   712→    <form onSubmit={submit} className="mt-3 space-y-3">
   713→      <div className="grid gap-2 sm:grid-cols-2">
   714→        <Input
   715→          required
   716→          placeholder="Full name *"
   717→          value={form.requesterName}
   718→          onChange={(e) => setForm({ ...form, requesterName: e.target.value })}
   719→        />
   720→        <Input
   721→          required
   722→          type="email"
   723→          placeholder="Work email *"
   724→          value={form.requesterEmail}
   725→          onChange={(e) => setForm({ ...form, requesterEmail: e.target.value })}
   726→        />
   727→      </div>
   728→      <Input
   729→        placeholder="Organization / clinic"
   730→        value={form.requesterOrg}
   731→        onChange={(e) => setForm({ ...form, requesterOrg: e.target.value })}
   732→      />
   733→      <div className="grid gap-2 sm:grid-cols-2">
   734→        <Input
   735→          placeholder="Quantity / scope"
   736→          value={form.quantity}
   737→          onChange={(e) => setForm({ ...form, quantity: e.target.value })}
   738→        />
   739→        <Textarea
   740→          rows={2}
   741→          placeholder="Notes (timeline, requirements)"
   742→          value={form.notes}
   743→          onChange={(e) => setForm({ ...form, notes: e.target.value })}
   744→          className="resize-none"
   745→        />
   746→      </div>
   747→      <label className="flex items-start gap-2 text-xs text-muted-foreground">
   748→        <input
   749→          type="checkbox"
   750→          checked={form.consent}
   751→          onChange={(e) => setForm({ ...form, consent: e.target.checked })}
   752→          className="mt-0.5"
   753→        />
   754→        <span>
   755→          I understand Novalyte AI is a technology platform that facilitates inquiries and does not
   756→          sell, warranty, or fulfill products directly.
   757→        </span>
   758→      </label>
   759→      <Button type="submit" className="w-full bg-teal-600 text-white hover:bg-teal-700" disabled={submitting}>
   760→        {submitting ? "Sending…" : <>Send quote request <ArrowRight className="ml-1 h-4 w-4" /></>}
   761→      </Button>
   762→    </form>
   763→  );
   764→}
